def records(path):
    """Validate every fragment checksum and read with rb, never r+b."""
    with path.open('rb') as stream:
        assert struct.unpack('<4sHB', stream.read(7)) == (b':W&B', 0xBEE1, 0)
        parts = []
        while True:
            remaining = 32768 - stream.tell() % 32768
            if remaining < 7:
                padding = stream.read(remaining)
                assert padding == b'\0' * remaining
            header = stream.read(7)
            if not header:
                assert not parts, 'Unfinished fragmented record'
                break
            assert len(header) == 7
            checksum, length, kind = struct.unpack('<IHB', header)
            assert kind in (1, 2, 3, 4)
            assert length <= 32768 - stream.tell() % 32768
            payload = stream.read(length)
            assert len(payload) == length
            assert zlib.crc32(payload, zlib.crc32(bytes([kind]))) & 0xffffffff == checksum
            if kind == 1:
                assert not parts
                yield Record.FromString(payload)
            elif kind == 2:
                assert not parts
                parts = [payload]
            else:
                assert parts
                parts.append(payload)
                if kind == 4:
                    yield Record.FromString(b''.join(parts))
                    parts = []


def key(item):
    return item.key or '/'.join(item.nested_key)


episodes = []
evaluations = []
config = {}
run_git_commit = None
counts = {}
total_steps = 0
allowed_config = {'algo', 'seed', 'total_timesteps', 'policy', 'encoder_mode', 'delay_mode', 'radius'}
allowed_metrics = {'train/episode_reward', 'train/episode_length', 'train/mean_episode_reward',
                   'train/mean_episode_length', '_timestamp',
                   'eval/step', 'eval/arrival_rate', 'eval/mean_reward'}
for record in records(RUN):
    kind = record.WhichOneof('record_type')
    counts[kind] = counts.get(kind, 0) + 1
    if kind == 'run':
        run_git_commit = record.run.git.commit
        for item in record.run.config.update:
            if key(item) in allowed_config:
                config[key(item)] = json.loads(item.value_json)
    elif kind == 'config':
        for item in record.config.update:
            if key(item) in allowed_config:
                config[key(item)] = json.loads(item.value_json)
    elif kind == 'history':
        values = {key(item): json.loads(item.value_json) for item in record.history.item
                  if key(item) in allowed_metrics}
        if 'train/episode_reward' in values:
            assert 'train/episode_length' in values
            length = int(values['train/episode_length'])
            assert length > 0 and math.isfinite(values['train/episode_reward'])
            total_steps += length
            episodes.append({'environment_step': total_steps, 'episode_length': length,
                             'episode_return': values['train/episode_reward'],
                             'logged_cumulative_mean_return': values['train/mean_episode_reward'],
                             'wall_time': values.get('_timestamp')})
        if 'eval/step' in values:
            evaluations.append({k: values[k] for k in
                                ('eval/step', 'eval/arrival_rate', 'eval/mean_reward')})
assert episodes and total_steps <= 10000000
with (ROOT/'original_episodes.csv').open('w', newline='') as stream:
    writer = csv.DictWriter(stream, fieldnames=list(episodes[0]))

def rolling(x, y):
    starts = np.searchsorted(x, x-50000, side='right')
    ends = np.arange(len(x))+1
    sums = np.concatenate(([0.], np.cumsum(y)))
    return (sums[ends]-sums[starts])/(ends-starts)


def describe(x, y, lo, hi):
    z = y[(x >= lo) & (x < hi)]
    return {'n': len(z), 'mean': float(np.mean(z)), 'median': float(np.median(z)),
            'min': float(np.min(z)), 'max': float(np.max(z))}

def draw(ax, x, y, color, limit):
    mask = x <= limit
    mean = rolling(x, y)
    ax.plot(x[mask]/1e6, y[mask], color=color, alpha=.2, linewidth=.6)
    ax.plot(x[mask]/1e6, mean[mask], color=color, linewidth=1.8)
    ax.grid(True, alpha=.25, linestyle='--', linewidth=.6)
    ax.set_xlim(0, limit/1e6)
